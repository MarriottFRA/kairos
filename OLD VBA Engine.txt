Function Validate() As Boolean
    'custom validaiton funciton (forces "end" on the first couple of tests as they are critical)
    Validate = True
    Dim Validation_Array As Variant
    
    Dim MaxRows As Double, adWorksheet As Worksheet
    Set adWorksheet = GetSheet_BaseName(ThisWorkbook, "AssociateDetails")
    MaxRows = adWorksheet.Cells(adWorksheet.Rows.count, 5).End(xlUp).Row
    
    'check row count
    If MaxRows < 6 Then
        MsgBox "Validaiton Error - Row Count" & vbCrLf _
            & "" & vbCrLf _
            & "The file does not work on files with 3 or less rows" & vbCrLf _
            & "" & vbCrLf _
            & "Please add extra rows", vbOKOnly Or vbExclamation, "Min Row Count!"
        Validate = False
        End
    End If
    

    'check structure
    Dim StructureIntegrity As Boolean: StructureIntegrity = True
    If adWorksheet.Cells(1, 45).Value2 = "Base_Percent_Increase" Then
    Else
        StructureIntegrity = False
    End If
    If adWorksheet.Range("FP1").Value2 = "ValueW_Social" Then
    Else
        StructureIntegrity = False
    End If
    If Not StructureIntegrity Then
        MsgBox "Validaiton Error - File Structre Issue" & vbCrLf _
            & "" & vbCrLf _
            & "The file has been optimised for a specific column structure, adding, deleting or moving columns will break calculations" & vbCrLf _
            & "" & vbCrLf _
            & "Please contact the support team to have this fixed", vbOKOnly Or vbExclamation, "File Structure Altered!"
        Validate = False
        End
    Else
    End If
    
    'pick up everything, will validate in sections and colour red.
    Dim i As Double
    Dim xColumn As Double
    Validation_Array = adWorksheet.Range("A1:FP" & MaxRows).Value2
    
    Dim xDepartments As New Dictionary
    Dim xTypePosition As New Dictionary
    Dim xPayroll As New Dictionary
    Dim xStats As New Dictionary
    Dim xAccount As New Dictionary
    Dim CustomColumnList As New Dictionary
    xDepartments.CompareMode = vbBinaryCompare
    xTypePosition.CompareMode = vbBinaryCompare
    xPayroll.CompareMode = vbBinaryCompare
    xStats.CompareMode = vbBinaryCompare
    xAccount.CompareMode = vbBinaryCompare
    CustomColumnList.CompareMode = vbTextCompare
    Dim ws_list As Worksheet
    Set ws_list = ThisWorkbook.Worksheets("Lists")

    ' Find last row for each column and populate dictionaries
    Dim lastRow As Long

    ' xTypePosition - Column 1
    lastRow = ws_list.Cells(ws_list.Rows.count, 1).End(xlUp).Row
    For i = 2 To lastRow
        If Not IsEmpty(ws_list.Cells(i, 1).Value2) Then
            xTypePosition(CStr(ws_list.Cells(i, 1).Value2)) = True
        End If
    Next i

    ' xDepartments - Column 2
    lastRow = ws_list.Cells(ws_list.Rows.count, 2).End(xlUp).Row
    For i = 2 To lastRow
        If Not IsEmpty(ws_list.Cells(i, 2).Value2) Then
            xDepartments(CStr(ws_list.Cells(i, 2).Value2)) = True
        End If
    Next i

    ' xPayroll - Column 8
    lastRow = ws_list.Cells(ws_list.Rows.count, 8).End(xlUp).Row
    For i = 2 To lastRow
        If Not IsEmpty(ws_list.Cells(i, 8).Value2) Then
            xPayroll(CStr(ws_list.Cells(i, 8).Value2)) = True
        End If
    Next i

    ' xStats - Column 11
    lastRow = ws_list.Cells(ws_list.Rows.count, 11).End(xlUp).Row
    For i = 2 To lastRow
        If Not IsEmpty(ws_list.Cells(i, 11).Value2) Then
            xStats(CStr(ws_list.Cells(i, 11).Value2)) = True
        End If
    Next i

    ' xAccount - Column 16
    lastRow = ws_list.Cells(ws_list.Rows.count, 16).End(xlUp).Row
    For i = 2 To lastRow
        If Not IsEmpty(ws_list.Cells(i, 16).Value2) Then
            xAccount(CStr(ws_list.Cells(i, 16).Value2)) = True
        End If
    Next i

    xAccount("Blank") = True
    
    With adWorksheet
    
        'check department
        For i = 4 To MaxRows
            For xColumn = 5 To 5
                If Not xDepartments.Exists(CStr(Validation_Array(i, xColumn))) Then
                    adWorksheet.Cells(i, xColumn).Interior.Color = RGB(255, 0, 0)
                    Validate = False
                End If
            Next xColumn
        Next i
        
        'check types
        For i = 4 To MaxRows
            For xColumn = 7 To 7
                If Not xTypePosition.Exists(CStr(Validation_Array(i, xColumn))) Then
                    adWorksheet.Cells(i, xColumn).Interior.Color = RGB(255, 0, 0)
                    Validate = False
                End If
            Next xColumn
        Next i
    
        'check entry stats
        For i = 4 To MaxRows
            For xColumn = 9 To 16
                If Not IsNumeric(Validation_Array(i, xColumn)) Then
                    adWorksheet.Cells(i, xColumn).Interior.Color = RGB(255, 0, 0)
                    Validate = False
                End If
            Next xColumn
        Next i
        
        'check next big batch
        For i = 4 To MaxRows
            For xColumn = 18 To 47
                If Not IsNumeric(Validation_Array(i, xColumn)) Then
                    adWorksheet.Cells(i, xColumn).Interior.Color = RGB(255, 0, 0)
                    Validate = False
                End If
            Next xColumn
        Next i
        
        'check hours accounts
        For i = 4 To MaxRows
            For xColumn = 48 To 49
                If Not xStats.Exists(CStr(Validation_Array(i, xColumn))) Then
                    adWorksheet.Cells(i, xColumn).Interior.Color = RGB(255, 0, 0)
                    Validate = False
                End If
            Next xColumn
        Next i
        
        'check FTE should be blank
        For i = 4 To MaxRows
            For xColumn = 50 To 50
                If Not Validation_Array(i, xColumn) = "Blank" Then
                    adWorksheet.Cells(i, xColumn).Interior.Color = RGB(255, 0, 0)
                    Validate = False
                End If
            Next xColumn
        Next i
        
        'check FTE should be blank
        For i = 4 To MaxRows
            For xColumn = 51 To 51
                If Not xPayroll.Exists(CStr(Validation_Array(i, xColumn))) Then
                    adWorksheet.Cells(i, xColumn).Interior.Color = RGB(255, 0, 0)
                    Validate = False
                End If
            Next xColumn
        Next i
        
        'next batch of numbers
        For i = 4 To MaxRows
            For xColumn = 52 To 55
                If Not IsNumeric(Validation_Array(i, xColumn)) Then
                    adWorksheet.Cells(i, xColumn).Interior.Color = RGB(255, 0, 0)
                    Validate = False
                End If
            Next xColumn
        Next i
        
        'check accrual account
        For i = 4 To MaxRows
            For xColumn = 56 To 56
                If Not xPayroll.Exists(CStr(Validation_Array(i, xColumn))) Then
                    adWorksheet.Cells(i, xColumn).Interior.Color = RGB(255, 0, 0)
                    Validate = False
                End If
            Next xColumn
        Next i
        
        'next batch of numbers
        For i = 4 To MaxRows
            For xColumn = 57 To 69
                If Not IsNumeric(Validation_Array(i, xColumn)) Then
                    adWorksheet.Cells(i, xColumn).Interior.Color = RGB(255, 0, 0)
                    Validate = False
                End If
            Next xColumn
        Next i
        
        'check accrual account
        For i = 4 To MaxRows
            For xColumn = 70 To 70
                If Not xPayroll.Exists(CStr(Validation_Array(i, xColumn))) Then
                    adWorksheet.Cells(i, xColumn).Interior.Color = RGB(255, 0, 0)
                    Validate = False
                End If
            Next xColumn
        Next i
        
        CustomColumnList(73) = True
        CustomColumnList(76) = True
        CustomColumnList(80) = True
        CustomColumnList(84) = True
        CustomColumnList(87) = True
        CustomColumnList(90) = True
        CustomColumnList(93) = True
        CustomColumnList(96) = True
        CustomColumnList(99) = True
        CustomColumnList(102) = True
        CustomColumnList(105) = True
        CustomColumnList(108) = True
        CustomColumnList(111) = True
        CustomColumnList(114) = True
        CustomColumnList(117) = True
        CustomColumnList(120) = True
        ' Overtime accounts
        CustomColumnList(126) = True
        CustomColumnList(127) = True
        CustomColumnList(128) = True
        CustomColumnList(141) = True
        CustomColumnList(154) = True
        CustomColumnList(171) = True

        
        
        For i = 4 To MaxRows
            For xColumn = 71 To 172
            
                If CustomColumnList(xColumn) Then
                    If Not xAccount.Exists(CStr(Validation_Array(i, xColumn))) Then
                        adWorksheet.Cells(i, xColumn).Interior.Color = RGB(255, 0, 0)
                        Validate = False
                    End If
                Else
                    If Not IsNumeric(Validation_Array(i, xColumn)) Then
                        adWorksheet.Cells(i, xColumn).Interior.Color = RGB(255, 0, 0)
                        Validate = False
                    End If
                End If
            Next xColumn
        Next i
        
        
    End With
    
    
    If Not Validate Then
        MsgBox "Validaiton Error - Data Entry Issues" & vbCrLf _
            & "" & vbCrLf _
            & "There are one or more data entry issues in the Associate Details tab, the invalid cells have been highlited in red." & vbCrLf _
            & "" & vbCrLf _
            & "Please fix the cells by entering the valid values for the fields.", vbOKOnly Or vbExclamation, "Data Entry Errors!"
    End If
    

End Function

Sub Generate_ExportData()
    ' ----------------------------------------------------------------
    ' Procedure Name: Generate_ExportData
    ' Purpose: Generate the main spread calculations, also updates the stats review,
    ' Purpose:   - The report generation should be seperated out in the next version to optimise the macro "flow"
    ' Procedure Kind: Sub
    ' Procedure Access: Public
    ' Author: Richard Porins (rpori625)
    ' Date: 28/07/2020
    ' ----------------------------------------------------------------
    'Declare Main Variables
    Dim MaxArrayLength As Double, adWorksheet As Worksheet, MaxRows As Double, MaxAccountColumns As Double, ComboArray() As Variant, FullDataArray() As Variant, WorkingColumn As Double
    Dim DepartmentColumn As Double, UniqueComboArray As Variant, UserSettingCalculation As Integer, TotalWorkingDays2 As Double, StatsReporting() As Variant, MaxStats As Double, statsTracker As Double
    Dim Base_Single_Month_Column As Double, Base_Percent_Increase_Column As Double, Base_Manual_Increase_Column As Double, Base_Increase_Month_Column As Double, Spread1_Column As Double
    Dim TotalValue As Double, ActiveMonthsIncrease As Double, IncreaseMonthly As Double, IncreaseMonth As Double, TotalWorkingMonths As Double, xSegment As Long, TotalWorkingDays As Double
    Dim Seasonality(13) As Double, xDivisor As Double, HourlyTest As Integer, StructureIntegrity As Boolean, Hourly_Manager_Check_Column As Double, TotalBaseSalary As Double, xMultiplayer As Double
    Dim TempColumn1 As Double, TempColumn2 As Double, TempColumn3 As Double, TempColumn4 As Double, TempColumn5 As Double, TempColumn6 As Double, TempColumn7 As Double, xDaysPerMonth(13, 2) As Double
    Dim OutputBudgetExport() As Variant, UniqueCounter As Double, CurrentLoader As String, SingleDayPay As Double, MaxRowsBuyOut As Double, UniqueStatsCombo As Variant, StatsCombo() As Variant, OutPutStatsCombo() As Variant
    'Declare throw away variables
    Dim RawData_Array As Variant, TempDataArray As Variant, Buyout_Array_Raw As Variant, UniqueTracker As Double, TempColumn8 As Double, Validation_Array As Variant
    Dim i As Double, o As Double, Found As Boolean, z As Double, x As Integer, P As Integer
    Dim CN_A1 As Integer, CN_A2 As Integer, CN_A3 As Integer, CN_A4 As Integer, CN_A5 As Integer, CN_A6 As Integer, CN_A7 As Integer, CN_A8 As Integer, CN_A9 As Integer, CN_A10 As Integer, CN_A11 As Integer, CN_A12 As Integer, CN_A13 As Integer, CN_A14 As Integer, CN_A15 As Integer, CN_A16 As Integer, CN_A17 As Integer, CN_A18 As Integer, CN_A19 As Integer, CN_A20 As Integer
    
    Dim ValidationTests As Boolean
    
    Calculate
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False
    UserSettingCalculation = CalculationSetting
    Application.Calculation = xlCalculationManual
    
    
    'fixed/dynamic amount for minor monthly flex, since this flexes hourly pay which never will be 100% accurate
    'First is weighted, second is flat for salaried spreads
    xDaysPerMonth(0, 0) = ThisWorkbook.Worksheets("Menu").Cells(11, 15).Value2
    xDaysPerMonth(0, 1) = 360
    TempDataArray = ThisWorkbook.Worksheets("Menu").Range("C11:N11") ' fixed as range should be immutable
    For i = 1 To 12
        xDaysPerMonth(i, 0) = TempDataArray(1, i)
        xDaysPerMonth(i, 1) = 30
    Next i
    '2D Array for data spread
    '0 - Base Salary
    '1 - Vacation Cost
    '2 - Pension
    '3 - Indemnity
    '4 - Stock Options
    '5 - Airfare
    '6 - Transportation
    '7 - Housing
    '8 - Food Allowance
    '9 - Hardship Allowance
    '10 - Custom Allowance 1
    '11 - Custom Allowance 2
    '12 - Custom Allowance 3
    '13 - Bonus Percentage
    '14 - Incentive Percentage 1
    '15 - Incentive Percentage 2
    '16 - Incentive Fixed Value 1
    '17 - Incentive Fixed Value 2
    '18 - Social Security/NI
    Dim MonthlyDataSpread(28, 13) As Double ' first number should match below
    MaxAccountColumns = 29 ' Fixed Account count based on amount of category columns to calculate hypothetical maximum
    'set base info
    Set adWorksheet = GetSheet_BaseName(ThisWorkbook, "AssociateDetails")
    MaxRows = adWorksheet.Cells(adWorksheet.Rows.count, 5).End(xlUp).Row
    
    Dim AllowedAccounts As New Collection
    
    AllowedAccounts.Add True, "Blank"
    'AllowedAccounts.Add True, "988110"
    'AllowedAccounts.Add True, "988110"

    
    'pre processing validaiton
    ValidationTests = Validate

    
    If Not ValidationTests Then
        Application.ScreenUpdating = True
        Application.EnableEvents = True
        Application.DisplayAlerts = True
        UserSettingCalculation = CalculationSetting
        Application.Calculation = UserSettingCalculation
        Exit Sub
    End If
    'end validaiton
        
    
    MaxRowsBuyOut = ThisWorkbook.Worksheets("Buyout & Manual Input").Cells(ThisWorkbook.Worksheets("Buyout & Manual Input").Rows.count, 2).End(xlUp).Row
    MaxArrayLength = (((MaxRows - 3) * MaxAccountColumns)) + (MaxRowsBuyOut * 2) - 5
    MaxStats = (MaxRows - 4)
    'since the file structure will be locked no point of dynamic pick up keep a manual data load, this will significantly speed up data processing
    'preload public varaibles that will need to be acccessed by other functions
    Base_Single_Month_Column = 31 'Total Salary 44
    Base_Percent_Increase_Column = Base_Single_Month_Column + 14
    Base_Manual_Increase_Column = Base_Single_Month_Column + 15
    Base_Increase_Month_Column = Base_Single_Month_Column + 16
    Spread1_Column = 18
    Hourly_Manager_Check_Column = 7
    RawData_Array = adWorksheet.Range("A4:FP" & MaxRows).Value2
    Buyout_Array_Raw = ThisWorkbook.Worksheets("Buyout & Manual Input").Range("A3:BD" & MaxRowsBuyOut).Value2
    
    'early binding for array to get unique value count.
    ReDim ComboArray(MaxArrayLength) As Variant
    ReDim FullDataArray(MaxArrayLength, 16) As Variant
    ReDim StatsReporting(MaxStats, 7) As Variant
    ReDim StatsCombo(MaxStats) As Variant
    DepartmentColumn = 5 'set department column
    xSegment = -1 'start segment at -1 so it itterates at increased increments
    statsTracker = -1
    'instead of looping though column data and wasting computational cycles to return modes of calculation the faster method is to
    'customise the calculation for each type and cycle only though the rows.
    'This method produces longer code but is hyperoptimised reducing cycle count by >200%
    CN_A1 = 0
    CN_A2 = 0
    CN_A3 = 0
    CN_A4 = 0
    CN_A5 = 0
    CN_A6 = 0
    CN_A7 = 0
    CN_A8 = 0
    CN_A9 = 0
    CN_A10 = 0
    CN_A11 = 0
    CN_A12 = 0
    CN_A13 = 0
    CN_A14 = 0
    CN_A15 = 0
    CN_A16 = 0
    CN_A17 = 0
    CN_A18 = 0
    CN_A19 = 0
    CN_A20 = 0
    'technically could set each variable to be the value rather than this if statement...
    If ThisWorkbook.Worksheets("Settings").Range("CN_A1").Value2 Then CN_A1 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A2").Value2 Then CN_A2 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A3").Value2 Then CN_A3 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A4").Value2 Then CN_A4 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A5").Value2 Then CN_A5 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A6").Value2 Then CN_A6 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A7").Value2 Then CN_A7 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A8").Value2 Then CN_A8 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A9").Value2 Then CN_A9 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A10").Value2 Then CN_A10 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A11").Value2 Then CN_A11 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A12").Value2 Then CN_A12 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A13").Value2 Then CN_A13 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A14").Value2 Then CN_A14 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A15").Value2 Then CN_A15 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A16").Value2 Then CN_A16 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A17").Value2 Then CN_A17 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A18").Value2 Then CN_A18 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A19").Value2 Then CN_A19 = 1
    If ThisWorkbook.Worksheets("Settings").Range("CN_A20").Value2 Then CN_A20 = 1
    
    For i = 1 To UBound(RawData_Array)
        '----------------------Section 1 - Basic Salary ---------------------------------
        xSegment = xSegment + 1
        'check if associate is hourly
        
        If Right(RawData_Array(i, Hourly_Manager_Check_Column), 10) = "Hourly Pay" Then
            HourlyTest = 0
        Else
            HourlyTest = 1
        End If
        
        WorkingColumn = 51 'account base column
        'set combos
        ComboArray(xSegment) = RawData_Array(i, DepartmentColumn) & RawData_Array(i, WorkingColumn)
        FullDataArray(xSegment, 0) = RawData_Array(i, DepartmentColumn) & RawData_Array(i, WorkingColumn)
        FullDataArray(xSegment, 13) = RawData_Array(i, DepartmentColumn + 2) ' Managed/Hourly
        FullDataArray(xSegment, 14) = RawData_Array(i, DepartmentColumn + 3) ' Job Title
        'wipe data from last loop
        TotalValue = 0
        ActiveMonthsIncrease = 0
        TotalWorkingMonths = 0
        TotalWorkingDays = 0
        TotalWorkingDays2 = 0
        TotalBaseSalary = 0
        IncreaseMonth = RawData_Array(i, Base_Increase_Month_Column)
        'preload seasonalty array so you don't need to reference the sheet
        For z = 1 To 12
            Seasonality(z) = RawData_Array(i, z + Spread1_Column - 1)
        Next z
        
        'precalculate base metadata for the loop
        For z = 1 To 12
            If Not Seasonality(z) = 0 Then
                TotalWorkingMonths = TotalWorkingMonths + Seasonality(z)
                TotalWorkingDays = TotalWorkingDays + (xDaysPerMonth(z, HourlyTest) * Seasonality(z))
                TotalWorkingDays2 = TotalWorkingDays2 + (xDaysPerMonth(z, 0) * Seasonality(z))
            Else
            End If
            If Not Seasonality(z) = 0 And Not z < IncreaseMonth Then ActiveMonthsIncrease = ActiveMonthsIncrease + Seasonality(z)
        Next z
        If ActiveMonthsIncrease = 0 Then
            IncreaseMonthly = 0
        Else
            IncreaseMonthly = RawData_Array(i, Base_Manual_Increase_Column) / ActiveMonthsIncrease
        End If
        'math loop for the base salary
        For z = 1 To 12
            If Not Seasonality(z) = 0 Then
                If z < IncreaseMonth Then
                    MonthlyDataSpread(0, z) = RawData_Array(i, Base_Single_Month_Column) * TotalWorkingMonths / TotalWorkingDays * xDaysPerMonth(z, HourlyTest) * Seasonality(z) + (RawData_Array(i, Base_Single_Month_Column + z) * Seasonality(z))
                Else
                    MonthlyDataSpread(0, z) = (RawData_Array(i, Base_Single_Month_Column) * TotalWorkingMonths / TotalWorkingDays * xDaysPerMonth(z, HourlyTest) * Seasonality(z) * (1 + RawData_Array(i, Base_Percent_Increase_Column))) + (IncreaseMonthly * Seasonality(z)) + (RawData_Array(i, Base_Single_Month_Column + z) * Seasonality(z))
                End If
            Else
                MonthlyDataSpread(0, z) = 0
            End If
            TotalBaseSalary = TotalBaseSalary + MonthlyDataSpread(0, z)
            FullDataArray(xSegment, z) = MonthlyDataSpread(0, z)
        Next z

        
        '----------------------Section 2 - Vacation ---------------------------------
        xSegment = xSegment + 1
        WorkingColumn = 70 ' holiday account
        'set combos
        ComboArray(xSegment) = RawData_Array(i, DepartmentColumn) & RawData_Array(i, WorkingColumn)
        FullDataArray(xSegment, 0) = RawData_Array(i, DepartmentColumn) & RawData_Array(i, WorkingColumn)
        FullDataArray(xSegment, 13) = FullDataArray(xSegment - 1, 13)
        FullDataArray(xSegment, 14) = FullDataArray(xSegment - 1, 14)
        TempColumn1 = RawData_Array(i, WorkingColumn - 17) 'holiday days for associate MODIFY
        TempColumn2 = 9 'Contract yearly days
        TempColumn2 = RawData_Array(i, TempColumn2) - RawData_Array(i, TempColumn2 + 1) - RawData_Array(i, TempColumn2 + 2)
        TempColumn3 = WorkingColumn - 13 'Monthly weights for holidays
        TempColumn4 = 0 ' cycler set to -1 for arithmetic purposes so first value is 0
        TempColumn7 = 0 ' Holiday arithmetic tracker, if total days exceeds tracker days remaining days set to 0
        SingleDayPay = RawData_Array(i, WorkingColumn - 16)
        'since total working months has be already affected by seasonality flex it would get a double impact here, so we need to adjust back the singledaypay
        If TotalWorkingMonths > 0 Then
            SingleDayPay = SingleDayPay / TotalWorkingMonths * 12
        End If

        For z = 1 To 12
            MonthlyDataSpread(1, z) = RawData_Array(i, TempColumn3 - 1 + z) * Seasonality(z) * TempColumn1
        Next z

        For z = 1 To 12
            If Not Seasonality(z) = 0 Then
                If z < IncreaseMonth Then
                    MonthlyDataSpread(1, z) = MonthlyDataSpread(1, z) * SingleDayPay
                Else
                    MonthlyDataSpread(1, z) = MonthlyDataSpread(1, z) * SingleDayPay * (1 + RawData_Array(i, Base_Percent_Increase_Column))
                End If
                
            Else
                MonthlyDataSpread(1, z) = 0
            End If
            FullDataArray(xSegment, z) = MonthlyDataSpread(1, z)
        Next z
        '----------------------Section 3 - Pension ---------------------------------
        'percentile spread on base salary
        xSegment = xSegment + 1
        WorkingColumn = 73 ' account pension
        FullDataArray(xSegment, 13) = FullDataArray(xSegment - 1, 13)
        FullDataArray(xSegment, 14) = FullDataArray(xSegment - 1, 14)
        ComboArray(xSegment) = RawData_Array(i, DepartmentColumn) & RawData_Array(i, WorkingColumn)
        FullDataArray(xSegment, 0) = RawData_Array(i, DepartmentColumn) & RawData_Array(i, WorkingColumn)
        
        TempColumn1 = RawData_Array(i, WorkingColumn - 1) 'pension % weight
        
        For z = 1 To 12
            MonthlyDataSpread(2, z) = MonthlyDataSpread(0, z) * TempColumn1
            FullDataArray(xSegment, z) = MonthlyDataSpread(2, z)
        Next z
        '----------------------Section 4 - Indemnity ---------------------------------
        'weighted monthly spread of total based on base salary
        xSegment = xSegment + 1
        WorkingColumn = 76 ' account Indemnity
        FullDataArray(xSegment, 13) = FullDataArray(xSegment - 1, 13)
        FullDataArray(xSegment, 14) = FullDataArray(xSegment - 1, 14)
        ComboArray(xSegment) = RawData_Array(i, DepartmentColumn) & RawData_Array(i, WorkingColumn)
        FullDataArray(xSegment, 0) = RawData_Array(i, DepartmentColumn) & RawData_Array(i, WorkingColumn)
        TempColumn1 = RawData_Array(i, WorkingColumn + 2) 'indemnity yearly value
        
        For z = 1 To 12
            If TempColumn1 = 0 Then
                MonthlyDataSpread(3, z) = 0
            Else
                MonthlyDataSpread(3, z) = TempColumn1 / TotalBaseSalary * MonthlyDataSpread(0, z)
            End If
            
            FullDataArray(xSegment, z) = MonthlyDataSpread(3, z)
        Next z
        '----------------------Section 5 - Stock Options ---------------------------------
        'Entierly Flat Spread by month
        xSegment = xSegment + 1
        WorkingColumn = 80 ' account Stock Options
        FullDataArray(xSegment, 13) = FullDataArray(xSegment - 1, 13)
        FullDataArray(xSegment, 14) = FullDataArray(xSegment - 1, 14)
        ComboArray(xSegment) = RawData_Array(i, DepartmentColumn) & RawData_Array(i, WorkingColumn)
        FullDataArray(xSegment, 0) = RawData_Array(i, DepartmentColumn) & RawData_Array(i, WorkingColumn)
        TempColumn1 = RawData_Array(i, WorkingColumn + 1) 'stock yearly value
        
        For z = 1 To 12
            If TempColumn1 = 0 Then
                MonthlyDataSpread(4, z) = 0
            Else
                If Not Seasonality(z) = 0 Then
                    MonthlyDataSpread(4, z) = TempColumn1 / TotalWorkingMonths * Seasonality(z)
                Else
                    MonthlyDataSpread(4, z) = 0
                End If
            End If
            FullDataArray(xSegment, z) = MonthlyDataSpread(4, z)
        Next z
        '----------------------Section 6 - AirfareTickets ---------------------------------
        'Entierly Flat Spread by month
        xSegment = xSegment + 1

Sub AddAllocationLines()
    Application.EnableEvents = False
    Dim BottomExport As Double, BottomAllocation As Double, Allocation As Variant, ExportS As Worksheet, AllocS As Worksheet, i As Double, x As Double, InsertRow As Double, AccountLoad As String
    Set ExportS = ThisWorkbook.Worksheets("Budget_Export")
    Set AllocS = ThisWorkbook.Worksheets("Allocations")
    BottomExport = ExportS.Cells(ExportS.Rows.count, 2).End(xlUp).Row
    BottomAllocation = AllocS.Cells(AllocS.Rows.count, 1).End(xlUp).Row
    If BottomAllocation = 9 Then Exit Sub
    Allocation = AllocS.Range("A10:K" & BottomAllocation).Value2
    InsertRow = BottomExport

    For i = 1 To 5
        For x = 1 To BottomAllocation - 9
            If Not AllocS.Cells(7, i * 2) = "Active" Then Exit For
            AccountLoad = AllocS.Cells(8, (i * 2) + 1).Value2
            InsertRow = InsertRow + 1
            ExportS.Cells(InsertRow, 1) = Allocation(x, 1) & "-" & AccountLoad
            ExportS.Cells(InsertRow, 2).NumberFormat = "@"
            ExportS.Cells(InsertRow, 2) = Allocation(x, 1)
            ExportS.Cells(InsertRow, 3) = AccountLoad
            ExportS.Cells(InsertRow, 5) = Allocation(x, i * 2)
            ExportS.Range(ExportS.Cells(InsertRow, 6), ExportS.Cells(InsertRow, 16)).Value2 = 0
        Next x
    Next i

    BottomExport = ExportS.Cells(ExportS.Rows.count, 2).End(xlUp).Row
    ExportS.Range("Q2:Q" & BottomExport).FormulaR1C1 = "=SUM(RC[-12]:RC[-1])"
    ExportS.Range("Q2:Q" & BottomExport).Calculate
    ExportS.Range("Q2:Q" & BottomExport).Value2 = ExportS.Range("Q2:Q" & BottomExport).Value2

End Sub

Sub ExportData2BST(AutoCreateAccounts As Boolean)
    'Main variables grouped by type
    Dim xLocation As String, xMessage As String
    Dim BST As Workbook, WB As Workbook
    Dim PasteSheet As Worksheet, ExportRaw As Variant
    Dim lastRow As Double, LogCounter As Double, CompleteCounter As Double, NewValue As Double, RowNumber As Double
    Dim AccCombo As Range
    Dim wasOpen As Boolean
    Dim Overwrite As Boolean: Overwrite = ThisWorkbook.Worksheets("Settings").Range("B6").Value2
    Dim UserSettingCalculation As Integer, MonthOffset As Integer
    're-usable throw away varaibles
    Dim i As Double, x As Integer
    'set static or starting variables
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False
    UserSettingCalculation = CalculationSetting
    Application.Calculation = xlCalculationManual
    
    
    
        'get all accounts and their codes
    Dim AccountMaxRows As Double
    Dim AccountsTable As Variant
    Set AccountList = New Dictionary
    
    
    With ThisWorkbook.Worksheets("Lists")
        AccountMaxRows = .Cells(.Rows.count, 16).End(xlUp).Row
        AccountsTable = .Range("P1:Q" & AccountMaxRows).Value2
    End With
    
    For i = 2 To AccountMaxRows
        AccountList(AccountsTable(i, 1)) = AccountsTable(i, 2)
    Next i
    
    
    
    MonthOffset = ThisWorkbook.Worksheets("Settings").Range("B7").Value2 - 1
    LogCounter = 0
    CompleteCounter = 0
    Set PasteSheet = ThisWorkbook.Worksheets("Log")
    PasteSheet.Range("A2:R99999").Clear
    With PasteSheet.Tab
        .ThemeColor = xlThemeColorAccent6
        .TintAndShade = 0.399975585192419
    End With

    xLocation = FileLocation("Select your BST", "*.xlsm") ' get BST Location
    If xLocation = "" Then
        MsgBox ("User Cancelled")
        Exit Sub
    Else
        For Each WB In Workbooks
            If WB.FullName = xLocation Then Set BST = WB 'Check if BST is Open
        Next WB
        If BST Is Nothing Then
            Set BST = Workbooks.Open(xLocation) 'set BST workbook
            wasOpen = False
        Else
            wasOpen = True
        End If
        
    End If
    With ThisWorkbook.Worksheets("Budget_Export")
        lastRow = .Cells(.Rows.count, "A").End(xlUp).Row
    End With
    For i = 2 To lastRow
        ThisWorkbook.Worksheets("Budget_Export").Cells(i, 4).Value2 = "Failed - See Log"
        With ThisWorkbook.Worksheets("Budget_Export").Cells(i, 4).Font
            .Color = -16777024
            .TintAndShade = 0
        End With
        'tries to select correct department
        If Not PasteSheet.Name = ThisWorkbook.Worksheets("Budget_Export").Cells(i, 2).Value2 Then
            If Not wsExists(BST, ThisWorkbook.Worksheets("Budget_Export").Cells(i, 2).Value2) Then GoTo NotFound
            Set PasteSheet = BST.Worksheets(ThisWorkbook.Worksheets("Budget_Export").Cells(i, 2).Value2)
        End If
        
        Set AccCombo = Nothing
        On Error Resume Next
        Set AccCombo = PasteSheet.Cells(Application.Match(ThisWorkbook.Worksheets("Budget_Export").Cells(i, 1).Value2, PasteSheet.Range("B:B"), 0), 2)
        On Error GoTo -1
        On Error GoTo 0
        
        If AccCombo Is Nothing Then
            If AutoCreateAccounts Then
                'call create new account
                Request_BST_New_Account ThisWorkbook.Worksheets("Budget_Export").Cells(i, 2).Value2, ThisWorkbook.Worksheets("Budget_Export").Cells(i, 3).Value2, "Payroll Auto Added", BST
                LogCounter = LogCounter + 1
                ThisWorkbook.Worksheets("Log").Range("A" & LogCounter + 1 & ":Q" & LogCounter + 1).Value2 = ThisWorkbook.Worksheets("Budget_Export").Range("A" & i & ":Q" & i).Value2
                ThisWorkbook.Worksheets("Log").Range("R" & LogCounter + 1).Value2 = "Accounted added to BST/FST."
                On Error GoTo AccNotFound
                Set AccCombo = PasteSheet.Cells(Application.Match(ThisWorkbook.Worksheets("Budget_Export").Cells(i, 1).Value2, PasteSheet.Range("B:B"), 0), 2)
                'the BST slows the code down, attempt to speed it back up
                Application.ScreenUpdating = False
                Application.EnableEvents = False
                Application.DisplayAlerts = False
                On Error GoTo -1
                On Error GoTo 0
            Else
                ThisWorkbook.Worksheets("Budget_Export").Cells(i, 4).Value2 = "Account not found"
                GoTo AccNotFound
            End If

        End If
        
        
        
        If Overwrite Then
            'overwrite
            Set AccCombo = AccCombo.Resize(, 12 - MonthOffset).Offset(0, 7 + MonthOffset)
            On Error GoTo Protected
            AccCombo.Value2 = ThisWorkbook.Worksheets("Budget_Export").Range(ThisWorkbook.Worksheets("Budget_Export").Cells(i, 5 + MonthOffset), ThisWorkbook.Worksheets("Budget_Export").Cells(i, 16)).Value2
            ThisWorkbook.Worksheets("Budget_Export").Cells(i, 4).Value2 = "Success"
            With ThisWorkbook.Worksheets("Budget_Export").Cells(i, 4).Font
                .ThemeColor = xlThemeColorAccent6
                .TintAndShade = -0.499984740745262
            End With
            On Error GoTo -1
            On Error GoTo 0
        Else
            'add
            Set AccCombo = AccCombo.Offset(0, 6 + MonthOffset)
            For x = 1 To 12 - MonthOffset
                Set AccCombo = AccCombo.Offset(0, 1)
                On Error GoTo Protected
                AccCombo.Value2 = AccCombo.Value2 + ThisWorkbook.Worksheets("Budget_Export").Cells(i, x + 4).Value2
                On Error GoTo -1
                On Error GoTo 0
            Next x
            ThisWorkbook.Worksheets("Budget_Export").Cells(i, 4).Value2 = "Success"
            With ThisWorkbook.Worksheets("Budget_Export").Cells(i, 4).Font
                .ThemeColor = xlThemeColorAccent6
                .TintAndShade = -0.499984740745262
            End With
        End If
        CompleteCounter = CompleteCounter + 1
        GoTo NextCycle
Protected:
        On Error GoTo -1
        On Error GoTo 0
        LogCounter = LogCounter + 1
        ThisWorkbook.Worksheets("Log").Range("A" & LogCounter + 1 & ":Q" & LogCounter + 1).Value2 = ThisWorkbook.Worksheets("Budget_Export").Range("A" & i & ":Q" & i).Value2
        ThisWorkbook.Worksheets("Log").Range("R" & LogCounter + 1).Value2 = "Combo Protected in BST - Please review, could be allocation."
        GoTo NextCycle
NotFound:
        On Error GoTo -1
        On Error GoTo 0
        LogCounter = LogCounter + 1
        ThisWorkbook.Worksheets("Log").Range("A" & LogCounter + 1 & ":Q" & LogCounter + 1).Value2 = ThisWorkbook.Worksheets("Budget_Export").Range("A" & i & ":Q" & i).Value2
        ThisWorkbook.Worksheets("Log").Range("R" & LogCounter + 1).Value2 = "Department Not Found - Please add department"
        GoTo NextCycle
AccNotFound:
        On Error GoTo -1
        On Error GoTo 0
        LogCounter = LogCounter + 1
        ThisWorkbook.Worksheets("Log").Range("A" & LogCounter + 1 & ":Q" & LogCounter + 1).Value2 = ThisWorkbook.Worksheets("Budget_Export").Range("A" & i & ":Q" & i).Value2
        ThisWorkbook.Worksheets("Log").Range("R" & LogCounter + 1).Value2 = "Account Not Found - Please add account"
NextCycle:
    Next i
    xMessage = "Injection Process Complete: " & CompleteCounter * 12 & " - Successfull Entries"
    If LogCounter > 0 Then
        xMessage = xMessage & " | " & LogCounter & " - Row Notification(s) please review 'Log' sheet"
        With ThisWorkbook.Worksheets("Log").Tab
            .Color = 255
            .TintAndShade = 0
        End With
    Else
    End If
    If Not wasOpen Then BST.Close savechanges:=True
    
    MsgBox (xMessage)
    
    Application.Calculation = CalculationSetting
    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
    Application.EnableEvents = True
    ThisWorkbook.Activate
    
End Sub

    Dim UserSettingCalculation As Variant
    Calculate
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False
    UserSettingCalculation = CalculationSetting
    Application.Calculation = xlCalculationManual
    'declare variables
    Dim rawData As Variant, lastRow As Double, wksh As Worksheet, wksh_source As Worksheet, BaseArray() As Variant, i As Double, x As Double, UniqueArray As Variant, CalculatedArray() As Variant

                ReDim CalculatedArray(1 To i - 1, 0 To 7)
                Exit For

        Next i
    Else
        ReDim Preserve UniqueArray(1 To UBound(rawData))
        ReDim CalculatedArray(1 To UBound(rawData), 0 To 7)
    End If
    
    SortedArray = QuickSortTable(rawData, -1, -1, 4)

    CurrentDep = ""
    CurrentRow = 0
    
    
    
    For i = 1 To UBound(SortedArray)
        If CurrentDep = SortedArray(i, 4) Then
        Else
            CurrentDep = SortedArray(i, 4)
            CurrentRow = CurrentRow + 1

        Else
        End If
    Next i
    
    Application.Calculation = CalculationSetting

Function ReturnCaseValue(x As String) As Double

    Select Case x
        Case "Base Salary(estimate)"
            ReturnCaseValue = 3
        Case "Head Count"
            ReturnCaseValue = 1
        Case "FTE Count"
            ReturnCaseValue = 2
        Case "Flat Spread"
            ReturnCaseValue = 7
        Case "Hours Worked"
            ReturnCaseValue = 4
        Case "Hours Paid"
            ReturnCaseValue = 5
        Case "Contract Days"
            ReturnCaseValue = 6
    End Select

End Function

                        TempColumn8 = TempColumn7 * RawData_Array(i, P + 1 + TempColumn3) + TempColumn8
                        TempColumn6 = TempColumn6 + TempColumn7
                        TempColumn7 = TempColumn7 - TempColumn7
                    Else

                            TempColumn5 = RawData_Array(i, P + TempColumn3) - TempColumn6
                            TempColumn6 = TempColumn6 + TempColumn5
                            TempColumn7 = TempColumn7 - TempColumn5
                            TempColumn8 = TempColumn5 * RawData_Array(i, P + 1 + TempColumn3) + TempColumn8
                            TempColumn3 = TempColumn3 + 2
                        Else
                            TempColumn6 = TempColumn6 + TempColumn7
                            TempColumn8 = TempColumn7 * RawData_Array(i, P + 1 + TempColumn3) + TempColumn8
                            TempColumn7 = TempColumn7 - TempColumn7
                        End If

Sub AddAllocationLines()
    Application.EnableEvents = False
    Dim BottomExport As Double, BottomAllocation As Double, Allocation As Variant, ExportS As Worksheet, AllocS As Worksheet, i As Double, x As Double, InsertRow As Double, AccountLoad As String
    Set ExportS = ThisWorkbook.Worksheets("Budget_Export")
    Set AllocS = ThisWorkbook.Worksheets("Allocations")
    BottomExport = ExportS.Cells(ExportS.Rows.count, 2).End(xlUp).Row
    BottomAllocation = AllocS.Cells(AllocS.Rows.count, 1).End(xlUp).Row
    If BottomAllocation = 9 Then Exit Sub
    Allocation = AllocS.Range("A10:K" & BottomAllocation).Value2
    InsertRow = BottomExport

    For i = 1 To 5
        For x = 1 To BottomAllocation - 9
            If Not AllocS.Cells(7, i * 2) = "Active" Then Exit For
            AccountLoad = AllocS.Cells(8, (i * 2) + 1).Value2
            InsertRow = InsertRow + 1
            ExportS.Cells(InsertRow, 1) = Allocation(x, 1) & "-" & AccountLoad
            ExportS.Cells(InsertRow, 2).NumberFormat = "@"
            ExportS.Cells(InsertRow, 2) = Allocation(x, 1)
            ExportS.Cells(InsertRow, 3) = AccountLoad
            ExportS.Cells(InsertRow, 5) = Allocation(x, i * 2)
            ExportS.Range(ExportS.Cells(InsertRow, 6), ExportS.Cells(InsertRow, 16)).Value2 = 0
        Next x
    Next i

    BottomExport = ExportS.Cells(ExportS.Rows.count, 2).End(xlUp).Row
    ExportS.Range("Q2:Q" & BottomExport).FormulaR1C1 = "=SUM(RC[-12]:RC[-1])"
    ExportS.Range("Q2:Q" & BottomExport).Calculate
    ExportS.Range("Q2:Q" & BottomExport).Value2 = ExportS.Range("Q2:Q" & BottomExport).Value2

End Sub

Sub ExportData2BST(AutoCreateAccounts As Boolean)
    'Main variables grouped by type
    Dim xLocation As String, xMessage As String
    Dim BST As Workbook, WB As Workbook
    Dim PasteSheet As Worksheet, ExportRaw As Variant
    Dim lastRow As Double, LogCounter As Double, CompleteCounter As Double, NewValue As Double, RowNumber As Double
    Dim AccCombo As Range
    Dim wasOpen As Boolean
    Dim Overwrite As Boolean: Overwrite = ThisWorkbook.Worksheets("Settings").Range("B6").Value2
    Dim UserSettingCalculation As Integer, MonthOffset As Integer
    're-usable throw away varaibles
    Dim i As Double, x As Integer
    'set static or starting variables
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False
    UserSettingCalculation = CalculationSetting
    Application.Calculation = xlCalculationManual
    
    
    
        'get all accounts and their codes
    Dim AccountMaxRows As Double
    Dim AccountsTable As Variant
    Set AccountList = New Dictionary
    
    
    With ThisWorkbook.Worksheets("Lists")
        AccountMaxRows = .Cells(.Rows.count, 16).End(xlUp).Row
        AccountsTable = .Range("P1:Q" & AccountMaxRows).Value2
    End With
    
    For i = 2 To AccountMaxRows
        AccountList(AccountsTable(i, 1)) = AccountsTable(i, 2)
    Next i
    
    
    
    MonthOffset = ThisWorkbook.Worksheets("Settings").Range("B7").Value2 - 1
    LogCounter = 0
    CompleteCounter = 0
    Set PasteSheet = ThisWorkbook.Worksheets("Log")
    PasteSheet.Range("A2:R99999").Clear
    With PasteSheet.Tab
        .ThemeColor = xlThemeColorAccent6
        .TintAndShade = 0.399975585192419
    End With

    xLocation = FileLocation("Select your BST", "*.xlsm") ' get BST Location
    If xLocation = "" Then
        MsgBox ("User Cancelled")
        Exit Sub
    Else
        For Each WB In Workbooks
            If WB.FullName = xLocation Then Set BST = WB 'Check if BST is Open
        Next WB
        If BST Is Nothing Then
            Set BST = Workbooks.Open(xLocation) 'set BST workbook
            wasOpen = False
        Else
            wasOpen = True
        End If
        
    End If
    With ThisWorkbook.Worksheets("Budget_Export")
        lastRow = .Cells(.Rows.count, "A").End(xlUp).Row
    End With
    For i = 2 To lastRow
        ThisWorkbook.Worksheets("Budget_Export").Cells(i, 4).Value2 = "Failed - See Log"
        With ThisWorkbook.Worksheets("Budget_Export").Cells(i, 4).Font
            .Color = -16777024
            .TintAndShade = 0
        End With
        'tries to select correct department
        If Not PasteSheet.Name = ThisWorkbook.Worksheets("Budget_Export").Cells(i, 2).Value2 Then
            If Not wsExists(BST, ThisWorkbook.Worksheets("Budget_Export").Cells(i, 2).Value2) Then GoTo NotFound
            Set PasteSheet = BST.Worksheets(ThisWorkbook.Worksheets("Budget_Export").Cells(i, 2).Value2)
        End If
        
        Set AccCombo = Nothing
        On Error Resume Next
        Set AccCombo = PasteSheet.Cells(Application.Match(ThisWorkbook.Worksheets("Budget_Export").Cells(i, 1).Value2, PasteSheet.Range("B:B"), 0), 2)
        On Error GoTo -1
        On Error GoTo 0
        
        If AccCombo Is Nothing Then
            If AutoCreateAccounts Then
                'call create new account
                Request_BST_New_Account ThisWorkbook.Worksheets("Budget_Export").Cells(i, 2).Value2, ThisWorkbook.Worksheets("Budget_Export").Cells(i, 3).Value2, "Payroll Auto Added", BST
                LogCounter = LogCounter + 1
                ThisWorkbook.Worksheets("Log").Range("A" & LogCounter + 1 & ":Q" & LogCounter + 1).Value2 = ThisWorkbook.Worksheets("Budget_Export").Range("A" & i & ":Q" & i).Value2
                ThisWorkbook.Worksheets("Log").Range("R" & LogCounter + 1).Value2 = "Accounted added to BST/FST."
                On Error GoTo AccNotFound
                Set AccCombo = PasteSheet.Cells(Application.Match(ThisWorkbook.Worksheets("Budget_Export").Cells(i, 1).Value2, PasteSheet.Range("B:B"), 0), 2)
                'the BST slows the code down, attempt to speed it back up
                Application.ScreenUpdating = False
                Application.EnableEvents = False
                Application.DisplayAlerts = False
                On Error GoTo -1
                On Error GoTo 0
            Else
                ThisWorkbook.Worksheets("Budget_Export").Cells(i, 4).Value2 = "Account not found"
                GoTo AccNotFound
            End If

        End If
        
        
        
        If Overwrite Then
            'overwrite
            Set AccCombo = AccCombo.Resize(, 12 - MonthOffset).Offset(0, 7 + MonthOffset)
            On Error GoTo Protected
            AccCombo.Value2 = ThisWorkbook.Worksheets("Budget_Export").Range(ThisWorkbook.Worksheets("Budget_Export").Cells(i, 5 + MonthOffset), ThisWorkbook.Worksheets("Budget_Export").Cells(i, 16)).Value2
            ThisWorkbook.Worksheets("Budget_Export").Cells(i, 4).Value2 = "Success"
            With ThisWorkbook.Worksheets("Budget_Export").Cells(i, 4).Font
                .ThemeColor = xlThemeColorAccent6
                .TintAndShade = -0.499984740745262
            End With
            On Error GoTo -1
            On Error GoTo 0
        Else
            'add
            Set AccCombo = AccCombo.Offset(0, 6 + MonthOffset)
            For x = 1 To 12 - MonthOffset
                Set AccCombo = AccCombo.Offset(0, 1)
                On Error GoTo Protected
                AccCombo.Value2 = AccCombo.Value2 + ThisWorkbook.Worksheets("Budget_Export").Cells(i, x + 4).Value2
                On Error GoTo -1
                On Error GoTo 0
            Next x
            ThisWorkbook.Worksheets("Budget_Export").Cells(i, 4).Value2 = "Success"
            With ThisWorkbook.Worksheets("Budget_Export").Cells(i, 4).Font
                .ThemeColor = xlThemeColorAccent6
                .TintAndShade = -0.499984740745262
            End With
        End If
        CompleteCounter = CompleteCounter + 1
        GoTo NextCycle
Protected:
        On Error GoTo -1
        On Error GoTo 0
        LogCounter = LogCounter + 1
        ThisWorkbook.Worksheets("Log").Range("A" & LogCounter + 1 & ":Q" & LogCounter + 1).Value2 = ThisWorkbook.Worksheets("Budget_Export").Range("A" & i & ":Q" & i).Value2
        ThisWorkbook.Worksheets("Log").Range("R" & LogCounter + 1).Value2 = "Combo Protected in BST - Please review, could be allocation."
        GoTo NextCycle
NotFound:
        On Error GoTo -1
        On Error GoTo 0
        LogCounter = LogCounter + 1
        ThisWorkbook.Worksheets("Log").Range("A" & LogCounter + 1 & ":Q" & LogCounter + 1).Value2 = ThisWorkbook.Worksheets("Budget_Export").Range("A" & i & ":Q" & i).Value2
        ThisWorkbook.Worksheets("Log").Range("R" & LogCounter + 1).Value2 = "Department Not Found - Please add department"
        GoTo NextCycle
AccNotFound:
        On Error GoTo -1
        On Error GoTo 0
        LogCounter = LogCounter + 1
        ThisWorkbook.Worksheets("Log").Range("A" & LogCounter + 1 & ":Q" & LogCounter + 1).Value2 = ThisWorkbook.Worksheets("Budget_Export").Range("A" & i & ":Q" & i).Value2
        ThisWorkbook.Worksheets("Log").Range("R" & LogCounter + 1).Value2 = "Account Not Found - Please add account"
NextCycle:
    Next i
    xMessage = "Injection Process Complete: " & CompleteCounter * 12 & " - Successfull Entries"
    If LogCounter > 0 Then
        xMessage = xMessage & " | " & LogCounter & " - Row Notification(s) please review 'Log' sheet"
        With ThisWorkbook.Worksheets("Log").Tab
            .Color = 255
            .TintAndShade = 0
        End With
    Else
    End If
    If Not wasOpen Then BST.Close savechanges:=True
    
    MsgBox (xMessage)
    
    Application.Calculation = CalculationSetting
    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
    Application.EnableEvents = True
    ThisWorkbook.Activate
    
End Sub

''
' Convert JSON string to object (Dictionary/Collection)
'
' @method ParseJson
' @param {String} json_String
' @return {Object} (Dictionary or Collection)
''
Public Function ParseJson(ByVal JsonString As String) As Object
    Dim json_Index As Long
    json_Index = 1

    ' Remove vbCr, vbLf, and vbTab from json_String
    JsonString = VBA.Replace(VBA.Replace(VBA.Replace(JsonString, VBA.vbCr, ""), VBA.vbLf, ""), VBA.vbTab, "")

    json_SkipSpaces JsonString, json_Index
    Select Case VBA.Mid$(JsonString, json_Index, 1)
    Case "{"
        Set ParseJson = json_ParseObject(JsonString, json_Index)
    Case "["
        Set ParseJson = json_ParseArray(JsonString, json_Index)
    Case Else
        ' Error: Invalid JSON string
        Err.Raise 10001, "JSONConverter", json_ParseErrorMessage(JsonString, json_Index, "Expecting '{' or '['")
    End Select
End Function

''
' Convert object (Dictionary/Collection/Array) to JSON
'
' @method ConvertToJson
' @param {Variant} JsonValue (Dictionary, Collection, or Array)
' @param {Integer|String} Whitespace "Pretty" print json with given number of spaces per indentation (Integer) or given string
' @return {String}
''
Public Function ConvertToJson(ByVal JsonValue As Variant, Optional ByVal Whitespace As Variant, Optional ByVal json_CurrentIndentation As Long = 0) As String
    Dim json_Buffer As String
    Dim json_BufferPosition As Long
    Dim json_BufferLength As Long
    Dim json_Index As Long
    Dim json_LBound As Long
    Dim json_UBound As Long
    Dim json_IsFirstItem As Boolean
    Dim json_Index2D As Long
    Dim json_LBound2D As Long
    Dim json_UBound2D As Long
    Dim json_IsFirstItem2D As Boolean
    Dim json_Key As Variant
    Dim json_Value As Variant
    Dim json_DateStr As String
    Dim json_Converted As String
    Dim json_SkipItem As Boolean
    Dim json_PrettyPrint As Boolean
    Dim json_Indentation As String
    Dim json_InnerIndentation As String

    json_LBound = -1
    json_UBound = -1
    json_IsFirstItem = True
    json_LBound2D = -1
    json_UBound2D = -1
    json_IsFirstItem2D = True
    json_PrettyPrint = Not IsMissing(Whitespace)

    Select Case VBA.VarType(JsonValue)
    Case VBA.vbNull
        ConvertToJson = "null"
    Case VBA.vbDate
        ' Date
        json_DateStr = ConvertToIso(VBA.CDate(JsonValue))

        ConvertToJson = """" & json_DateStr & """"
    Case VBA.vbString
        ' String (or large number encoded as string)
        If Not JsonOptions.UseDoubleForLargeNumbers And json_StringIsLargeNumber(JsonValue) Then
            ConvertToJson = JsonValue
        Else
            ConvertToJson = """" & json_Encode(JsonValue) & """"
        End If
    Case VBA.vbBoolean
        If JsonValue Then
            ConvertToJson = "true"
        Else
            ConvertToJson = "false"
        End If
    Case VBA.vbArray To VBA.vbArray + VBA.vbByte
        If json_PrettyPrint Then
            If VBA.VarType(Whitespace) = VBA.vbString Then
                json_Indentation = VBA.String$(json_CurrentIndentation + 1, Whitespace)
                json_InnerIndentation = VBA.String$(json_CurrentIndentation + 2, Whitespace)
            Else
                json_Indentation = VBA.Space$((json_CurrentIndentation + 1) * Whitespace)
                json_InnerIndentation = VBA.Space$((json_CurrentIndentation + 2) * Whitespace)
            End If
        End If

        ' Array
        json_BufferAppend json_Buffer, "[", json_BufferPosition, json_BufferLength

        On Error Resume Next

        json_LBound = LBound(JsonValue, 1)
        json_UBound = UBound(JsonValue, 1)
        json_LBound2D = LBound(JsonValue, 2)
        json_UBound2D = UBound(JsonValue, 2)

        If json_LBound >= 0 And json_UBound >= 0 Then
            For json_Index = json_LBound To json_UBound
                If json_IsFirstItem Then
                    json_IsFirstItem = False
                Else
                    ' Append comma to previous line
                    json_BufferAppend json_Buffer, ",", json_BufferPosition, json_BufferLength
                End If

                If json_LBound2D >= 0 And json_UBound2D >= 0 Then
                    ' 2D Array
                    If json_PrettyPrint Then
                        json_BufferAppend json_Buffer, vbNewLine, json_BufferPosition, json_BufferLength
                    End If
                    json_BufferAppend json_Buffer, json_Indentation & "[", json_BufferPosition, json_BufferLength

                    For json_Index2D = json_LBound2D To json_UBound2D
                        If json_IsFirstItem2D Then
                            json_IsFirstItem2D = False
                        Else
                            json_BufferAppend json_Buffer, ",", json_BufferPosition, json_BufferLength
                        End If

                        json_Converted = ConvertToJson(JsonValue(json_Index, json_Index2D), Whitespace, json_CurrentIndentation + 2)

                        ' For Arrays/Collections, undefined (Empty/Nothing) is treated as null
                        If json_Converted = "" Then
                            ' (nest to only check if converted = "")
                            If json_IsUndefined(JsonValue(json_Index, json_Index2D)) Then
                                json_Converted = "null"
                            End If
                        End If

                        If json_PrettyPrint Then
                            json_Converted = vbNewLine & json_InnerIndentation & json_Converted
                        End If

                        json_BufferAppend json_Buffer, json_Converted, json_BufferPosition, json_BufferLength
                    Next json_Index2D

                    If json_PrettyPrint Then
                        json_BufferAppend json_Buffer, vbNewLine, json_BufferPosition, json_BufferLength
                    End If

                    json_BufferAppend json_Buffer, json_Indentation & "]", json_BufferPosition, json_BufferLength
                    json_IsFirstItem2D = True
                Else
                    ' 1D Array
                    json_Converted = ConvertToJson(JsonValue(json_Index), Whitespace, json_CurrentIndentation + 1)

                    ' For Arrays/Collections, undefined (Empty/Nothing) is treated as null
                    If json_Converted = "" Then
                        ' (nest to only check if converted = "")
                        If json_IsUndefined(JsonValue(json_Index)) Then
                            json_Converted = "null"
                        End If
                    End If

                    If json_PrettyPrint Then
                        json_Converted = vbNewLine & json_Indentation & json_Converted
                    End If

                    json_BufferAppend json_Buffer, json_Converted, json_BufferPosition, json_BufferLength
                End If
            Next json_Index
        End If

        On Error GoTo 0

        If json_PrettyPrint Then
            json_BufferAppend json_Buffer, vbNewLine, json_BufferPosition, json_BufferLength

            If VBA.VarType(Whitespace) = VBA.vbString Then
                json_Indentation = VBA.String$(json_CurrentIndentation, Whitespace)
            Else
                json_Indentation = VBA.Space$(json_CurrentIndentation * Whitespace)
            End If
        End If

        json_BufferAppend json_Buffer, json_Indentation & "]", json_BufferPosition, json_BufferLength

        ConvertToJson = json_BufferToString(json_Buffer, json_BufferPosition)

    ' Dictionary or Collection
    Case VBA.vbObject
        If json_PrettyPrint Then
            If VBA.VarType(Whitespace) = VBA.vbString Then
                json_Indentation = VBA.String$(json_CurrentIndentation + 1, Whitespace)
            Else
                json_Indentation = VBA.Space$((json_CurrentIndentation + 1) * Whitespace)
            End If
        End If

        ' Dictionary
        If VBA.TypeName(JsonValue) = "Dictionary" Then
            json_BufferAppend json_Buffer, "{", json_BufferPosition, json_BufferLength
            For Each json_Key In JsonValue.Keys
                ' For Objects, undefined (Empty/Nothing) is not added to object
                json_Converted = ConvertToJson(JsonValue(json_Key), Whitespace, json_CurrentIndentation + 1)
                If json_Converted = "" Then
                    json_SkipItem = json_IsUndefined(JsonValue(json_Key))
                Else
                    json_SkipItem = False
                End If

                If Not json_SkipItem Then
                    If json_IsFirstItem Then
                        json_IsFirstItem = False
                    Else
                        json_BufferAppend json_Buffer, ",", json_BufferPosition, json_BufferLength
                    End If

                    If json_PrettyPrint Then
                        json_Converted = vbNewLine & json_Indentation & """" & json_Key & """: " & json_Converted
                    Else
                        json_Converted = """" & json_Key & """:" & json_Converted
                    End If

                    json_BufferAppend json_Buffer, json_Converted, json_BufferPosition, json_BufferLength
                End If
            Next json_Key

            If json_PrettyPrint Then
                json_BufferAppend json_Buffer, vbNewLine, json_BufferPosition, json_BufferLength

                If VBA.VarType(Whitespace) = VBA.vbString Then
                    json_Indentation = VBA.String$(json_CurrentIndentation, Whitespace)
                Else
                    json_Indentation = VBA.Space$(json_CurrentIndentation * Whitespace)
                End If
            End If

            json_BufferAppend json_Buffer, json_Indentation & "}", json_BufferPosition, json_BufferLength

        ' Collection
        ElseIf VBA.TypeName(JsonValue) = "Collection" Then
            json_BufferAppend json_Buffer, "[", json_BufferPosition, json_BufferLength
            For Each json_Value In JsonValue
                If json_IsFirstItem Then
                    json_IsFirstItem = False
                Else
                    json_BufferAppend json_Buffer, ",", json_BufferPosition, json_BufferLength
                End If

                json_Converted = ConvertToJson(json_Value, Whitespace, json_CurrentIndentation + 1)

                ' For Arrays/Collections, undefined (Empty/Nothing) is treated as null
                If json_Converted = "" Then
                    ' (nest to only check if converted = "")
                    If json_IsUndefined(json_Value) Then
                        json_Converted = "null"
                    End If
                End If

                If json_PrettyPrint Then
                    json_Converted = vbNewLine & json_Indentation & json_Converted
                End If

                json_BufferAppend json_Buffer, json_Converted, json_BufferPosition, json_BufferLength
            Next json_Value

            If json_PrettyPrint Then
                json_BufferAppend json_Buffer, vbNewLine, json_BufferPosition, json_BufferLength

                If VBA.VarType(Whitespace) = VBA.vbString Then
                    json_Indentation = VBA.String$(json_CurrentIndentation, Whitespace)
                Else
                    json_Indentation = VBA.Space$(json_CurrentIndentation * Whitespace)
                End If
            End If

            json_BufferAppend json_Buffer, json_Indentation & "]", json_BufferPosition, json_BufferLength
        End If

        ConvertToJson = json_BufferToString(json_Buffer, json_BufferPosition)
    Case VBA.vbInteger, VBA.vbLong, VBA.vbSingle, VBA.vbDouble, VBA.vbCurrency, VBA.vbDecimal
        ' Number (use decimals for numbers)
        ConvertToJson = VBA.Replace(JsonValue, ",", ".")
    Case Else
        ' vbEmpty, vbError, vbDataObject, vbByte, vbUserDefinedType
        ' Use VBA's built-in to-string
        On Error Resume Next
        ConvertToJson = JsonValue
        On Error GoTo 0
    End Select
End Function

Private Function json_ParseObject(json_String As String, ByRef json_Index As Long) As Dictionary
'are you getting a user defined error here? Oh my? Could it be because you have not enabled 'Microsoft Scripting Runtime' references?


    Dim json_Key As String
    Dim json_NextChar As String

    Set json_ParseObject = New Dictionary
    json_SkipSpaces json_String, json_Index
    If VBA.Mid$(json_String, json_Index, 1) <> "{" Then
        Err.Raise 10001, "JSONConverter", json_ParseErrorMessage(json_String, json_Index, "Expecting '{'")
    Else
        json_Index = json_Index + 1

        Do
            json_SkipSpaces json_String, json_Index
            If VBA.Mid$(json_String, json_Index, 1) = "}" Then
                json_Index = json_Index + 1
                Exit Function
            ElseIf VBA.Mid$(json_String, json_Index, 1) = "," Then
                json_Index = json_Index + 1
                json_SkipSpaces json_String, json_Index
            End If

            json_Key = json_ParseKey(json_String, json_Index)
            json_NextChar = json_Peek(json_String, json_Index)
            If json_NextChar = "[" Or json_NextChar = "{" Then
                Set json_ParseObject.Item(json_Key) = json_ParseValue(json_String, json_Index)
            Else
                json_ParseObject.Item(json_Key) = json_ParseValue(json_String, json_Index)
            End If
        Loop
    End If
End Function

Private Function json_ParseArray(json_String As String, ByRef json_Index As Long) As Collection
    Set json_ParseArray = New Collection

    json_SkipSpaces json_String, json_Index
    If VBA.Mid$(json_String, json_Index, 1) <> "[" Then
        Err.Raise 10001, "JSONConverter", json_ParseErrorMessage(json_String, json_Index, "Expecting '['")
    Else
        json_Index = json_Index + 1

        Do
            json_SkipSpaces json_String, json_Index
            If VBA.Mid$(json_String, json_Index, 1) = "]" Then
                json_Index = json_Index + 1
                Exit Function
            ElseIf VBA.Mid$(json_String, json_Index, 1) = "," Then
                json_Index = json_Index + 1
                json_SkipSpaces json_String, json_Index
            End If

            json_ParseArray.Add json_ParseValue(json_String, json_Index)
        Loop
    End If
End Function

Private Function json_ParseValue(json_String As String, ByRef json_Index As Long) As Variant
    json_SkipSpaces json_String, json_Index
    Select Case VBA.Mid$(json_String, json_Index, 1)
    Case "{"
        Set json_ParseValue = json_ParseObject(json_String, json_Index)
    Case "["
        Set json_ParseValue = json_ParseArray(json_String, json_Index)
    Case """", "'"
        json_ParseValue = json_ParseString(json_String, json_Index)
    Case Else
        If VBA.Mid$(json_String, json_Index, 4) = "true" Then
            json_ParseValue = True
            json_Index = json_Index + 4
        ElseIf VBA.Mid$(json_String, json_Index, 5) = "false" Then
            json_ParseValue = False
            json_Index = json_Index + 5
        ElseIf VBA.Mid$(json_String, json_Index, 4) = "null" Then
            json_ParseValue = Null
            json_Index = json_Index + 4
        ElseIf VBA.InStr("+-0123456789", VBA.Mid$(json_String, json_Index, 1)) Then
            json_ParseValue = json_ParseNumber(json_String, json_Index)
        Else
            Err.Raise 10001, "JSONConverter", json_ParseErrorMessage(json_String, json_Index, "Expecting 'STRING', 'NUMBER', null, true, false, '{', or '['")
        End If
    End Select
End Function

Private Function json_ParseString(json_String As String, ByRef json_Index As Long) As String
    Dim json_Quote As String
    Dim json_Char As String
    Dim json_Code As String
    Dim json_Buffer As String
    Dim json_BufferPosition As Long
    Dim json_BufferLength As Long

    json_SkipSpaces json_String, json_Index

    ' Store opening quote to look for matching closing quote
    json_Quote = VBA.Mid$(json_String, json_Index, 1)
    json_Index = json_Index + 1

    Do While json_Index > 0 And json_Index <= Len(json_String)
        json_Char = VBA.Mid$(json_String, json_Index, 1)

        Select Case json_Char
        Case "\"
            ' Escaped string, \\, or \/
            json_Index = json_Index + 1
            json_Char = VBA.Mid$(json_String, json_Index, 1)

            Select Case json_Char
            Case """", "\", "/", "'"
                json_BufferAppend json_Buffer, json_Char, json_BufferPosition, json_BufferLength
                json_Index = json_Index + 1
            Case "b"
                json_BufferAppend json_Buffer, vbBack, json_BufferPosition, json_BufferLength
                json_Index = json_Index + 1
            Case "f"
                json_BufferAppend json_Buffer, vbFormFeed, json_BufferPosition, json_BufferLength
                json_Index = json_Index + 1
            Case "n"
                json_BufferAppend json_Buffer, vbCrLf, json_BufferPosition, json_BufferLength
                json_Index = json_Index + 1
            Case "r"
                json_BufferAppend json_Buffer, vbCr, json_BufferPosition, json_BufferLength
                json_Index = json_Index + 1
            Case "t"
                json_BufferAppend json_Buffer, vbTab, json_BufferPosition, json_BufferLength
                json_Index = json_Index + 1
            Case "u"
                ' Unicode character escape (e.g. \u00a9 = Copyright)
                json_Index = json_Index + 1
                json_Code = VBA.Mid$(json_String, json_Index, 4)
                json_BufferAppend json_Buffer, VBA.ChrW(VBA.Val("&h" + json_Code)), json_BufferPosition, json_BufferLength
                json_Index = json_Index + 4
            End Select
        Case json_Quote
            json_ParseString = json_BufferToString(json_Buffer, json_BufferPosition)
            json_Index = json_Index + 1
            Exit Function
        Case Else
            json_BufferAppend json_Buffer, json_Char, json_BufferPosition, json_BufferLength
            json_Index = json_Index + 1
        End Select
    Loop
End Function

Private Function json_ParseNumber(json_String As String, ByRef json_Index As Long) As Variant
    Dim json_Char As String
    Dim json_Value As String
    Dim json_IsLargeNumber As Boolean

    json_SkipSpaces json_String, json_Index

    Do While json_Index > 0 And json_Index <= Len(json_String)
        json_Char = VBA.Mid$(json_String, json_Index, 1)

        If VBA.InStr("+-0123456789.eE", json_Char) Then
            ' Unlikely to have massive number, so use simple append rather than buffer here
            json_Value = json_Value & json_Char
            json_Index = json_Index + 1
        Else
            ' Excel only stores 15 significant digits, so any numbers larger than that are truncated
            ' This can lead to issues when BIGINT's are used (e.g. for Ids or Credit Cards), as they will be invalid above 15 digits
            ' See: http://support.microsoft.com/kb/269370
            '
            ' Fix: Parse -> String, Convert -> String longer than 15/16 characters containing only numbers and decimal points -> Number
            ' (decimal doesn't factor into significant digit count, so if present check for 15 digits + decimal = 16)
            json_IsLargeNumber = IIf(InStr(json_Value, "."), Len(json_Value) >= 17, Len(json_Value) >= 16)
            If Not JsonOptions.UseDoubleForLargeNumbers And json_IsLargeNumber Then
                json_ParseNumber = json_Value
            Else
                ' VBA.Val does not use regional settings, so guard for comma is not needed
                json_ParseNumber = VBA.Val(json_Value)
            End If
            Exit Function
        End If
    Loop
End Function

Private Function json_ParseKey(json_String As String, ByRef json_Index As Long) As String
    ' Parse key with single or double quotes
    If VBA.Mid$(json_String, json_Index, 1) = """" Or VBA.Mid$(json_String, json_Index, 1) = "'" Then
        json_ParseKey = json_ParseString(json_String, json_Index)
    ElseIf JsonOptions.AllowUnquotedKeys Then
        Dim json_Char As String
        Do While json_Index > 0 And json_Index <= Len(json_String)
            json_Char = VBA.Mid$(json_String, json_Index, 1)
            If (json_Char <> " ") And (json_Char <> ":") Then
                json_ParseKey = json_ParseKey & json_Char
                json_Index = json_Index + 1
            Else
                Exit Do
            End If
        Loop
    Else
        Err.Raise 10001, "JSONConverter", json_ParseErrorMessage(json_String, json_Index, "Expecting '""' or '''")
    End If

    ' Check for colon and skip if present or throw if not present
    json_SkipSpaces json_String, json_Index
    If VBA.Mid$(json_String, json_Index, 1) <> ":" Then
        Err.Raise 10001, "JSONConverter", json_ParseErrorMessage(json_String, json_Index, "Expecting ':'")
    Else
        json_Index = json_Index + 1
    End If
End Function

Private Function json_IsUndefined(ByVal json_Value As Variant) As Boolean
    ' Empty / Nothing -> undefined
    Select Case VBA.VarType(json_Value)
    Case VBA.vbEmpty
        json_IsUndefined = True
    Case VBA.vbObject
        Select Case VBA.TypeName(json_Value)
        Case "Empty", "Nothing"
            json_IsUndefined = True
        End Select
    End Select
End Function

Private Function json_Encode(ByVal json_Text As Variant) As String
    ' Reference: http://www.ietf.org/rfc/rfc4627.txt
    ' Escape: ", \, /, backspace, form feed, line feed, carriage return, tab
    Dim json_Index As Long
    Dim json_Char As String
    Dim json_AscCode As Long
    Dim json_Buffer As String
    Dim json_BufferPosition As Long
    Dim json_BufferLength As Long

    For json_Index = 1 To VBA.Len(json_Text)
        json_Char = VBA.Mid$(json_Text, json_Index, 1)
        json_AscCode = VBA.AscW(json_Char)

        ' When AscW returns a negative number, it returns the twos complement form of that number.
        ' To convert the twos complement notation into normal binary notation, add 0xFFF to the return result.
        ' https://support.microsoft.com/en-us/kb/272138
        If json_AscCode < 0 Then
            json_AscCode = json_AscCode + 65536
        End If

        ' From spec, ", \, and control characters must be escaped (solidus is optional)

        Select Case json_AscCode
        Case 34
            ' " -> 34 -> \"
            json_Char = "\"""
        Case 92
            ' \ -> 92 -> \\
            json_Char = "\\"
        Case 47
            ' / -> 47 -> \/ (optional)
            If JsonOptions.EscapeSolidus Then
                json_Char = "\/"
            End If
        Case 8
            ' backspace -> 8 -> \b
            json_Char = "\b"
        Case 12
            ' form feed -> 12 -> \f
            json_Char = "\f"
        Case 10
            ' line feed -> 10 -> \n
            json_Char = "\n"
        Case 13
            ' carriage return -> 13 -> \r
            json_Char = "\r"
        Case 9
            ' tab -> 9 -> \t
            json_Char = "\t"
        Case 0 To 31, 127 To 65535
            ' Non-ascii characters -> convert to 4-digit hex
            json_Char = "\u" & VBA.Right$("0000" & VBA.Hex$(json_AscCode), 4)
        End Select

        json_BufferAppend json_Buffer, json_Char, json_BufferPosition, json_BufferLength
    Next json_Index

    json_Encode = json_BufferToString(json_Buffer, json_BufferPosition)
End Function

Private Function json_Peek(json_String As String, ByVal json_Index As Long, Optional json_NumberOfCharacters As Long = 1) As String
    ' "Peek" at the next number of characters without incrementing json_Index (ByVal instead of ByRef)
    json_SkipSpaces json_String, json_Index
    json_Peek = VBA.Mid$(json_String, json_Index, json_NumberOfCharacters)
End Function

Private Sub json_SkipSpaces(json_String As String, ByRef json_Index As Long)
    ' Increment index to skip over spaces
    Do While json_Index > 0 And json_Index <= VBA.Len(json_String) And VBA.Mid$(json_String, json_Index, 1) = " "
        json_Index = json_Index + 1
    Loop
End Sub

Private Function json_StringIsLargeNumber(json_String As Variant) As Boolean
    ' Check if the given string is considered a "large number"
    ' (See json_ParseNumber)

    Dim json_Length As Long
    Dim json_CharIndex As Long
    json_Length = VBA.Len(json_String)

    ' Length with be at least 16 characters and assume will be less than 100 characters
    If json_Length >= 16 And json_Length <= 100 Then
        Dim json_CharCode As String

        json_StringIsLargeNumber = True

        For json_CharIndex = 1 To json_Length
            json_CharCode = VBA.Asc(VBA.Mid$(json_String, json_CharIndex, 1))
            Select Case json_CharCode
            ' Look for .|0-9|E|e
            Case 46, 48 To 57, 69, 101
                ' Continue through characters
            Case Else
                json_StringIsLargeNumber = False
                Exit Function
            End Select
        Next json_CharIndex
    End If
End Function

Private Function json_ParseErrorMessage(json_String As String, ByRef json_Index As Long, ErrorMessage As String)
    ' Provide detailed parse error message, including details of where and what occurred
    '
    ' Example:
    ' Error parsing JSON:
    ' {"abcde":True}
    '          ^
    ' Expecting 'STRING', 'NUMBER', null, true, false, '{', or '['

    Dim json_StartIndex As Long
    Dim json_StopIndex As Long

    ' Include 10 characters before and after error (if possible)
    json_StartIndex = json_Index - 10
    json_StopIndex = json_Index + 10
    If json_StartIndex <= 0 Then
        json_StartIndex = 1
    End If
    If json_StopIndex > VBA.Len(json_String) Then
        json_StopIndex = VBA.Len(json_String)
    End If

    json_ParseErrorMessage = "Error parsing JSON:" & VBA.vbNewLine & VBA.Mid$(json_String, json_StartIndex, json_StopIndex - json_StartIndex + 1) & VBA.vbNewLine & VBA.Space$(json_Index - json_StartIndex) & "^" & VBA.vbNewLine & ErrorMessage
End Function

Private Sub json_BufferAppend(ByRef json_Buffer As String, ByRef json_Append As Variant, ByRef json_BufferPosition As Long, ByRef json_BufferLength As Long)
    ' VBA can be slow to append strings due to allocating a new string for each append
    ' Instead of using the traditional append, allocate a large empty string and then copy string at append position
    '
    ' Example:
    ' Buffer: "abc  "
    ' Append: "def"
    ' Buffer Position: 3
    ' Buffer Length: 5
    '
    ' Buffer position + Append length > Buffer length -> Append chunk of blank space to buffer
    ' Buffer: "abc       "
    ' Buffer Length: 10
    '
    ' Put "def" into buffer at position 3 (0-based)
    ' Buffer: "abcdef    "
    '
    ' Approach based on cStringBuilder from vbAccelerator
    ' http://www.vbaccelerator.com/home/VB/Code/Techniques/RunTime_Debug_Tracing/VB6_Tracer_Utility_zip_cStringBuilder_cls.asp
    '
    ' and clsStringAppend from Philip Swannell
    ' https://github.com/VBA-tools/VBA-JSON/pull/82

    Dim json_AppendLength As Long
    Dim json_LengthPlusPosition As Long

    json_AppendLength = VBA.Len(json_Append)
    json_LengthPlusPosition = json_AppendLength + json_BufferPosition

    If json_LengthPlusPosition > json_BufferLength Then
        ' Appending would overflow buffer, add chunk
        ' (double buffer length or append length, whichever is bigger)
        Dim json_AddedLength As Long
        json_AddedLength = IIf(json_AppendLength > json_BufferLength, json_AppendLength, json_BufferLength)

        json_Buffer = json_Buffer & VBA.Space$(json_AddedLength)
        json_BufferLength = json_BufferLength + json_AddedLength
    End If

    ' Note: Namespacing with VBA.Mid$ doesn't work properly here, throwing compile error:
    ' Function call on left-hand side of assignment must return Variant or Object
    Mid$(json_Buffer, json_BufferPosition + 1, json_AppendLength) = CStr(json_Append)
    json_BufferPosition = json_BufferPosition + json_AppendLength
End Sub

Private Function json_BufferToString(ByRef json_Buffer As String, ByVal json_BufferPosition As Long) As String
    If json_BufferPosition > 0 Then
        json_BufferToString = VBA.Left$(json_Buffer, json_BufferPosition)
    End If
End Function

''
' Parse ISO 8601 date string to local date
'
' @method ParseIso
' @param {Date} utc_IsoString
' @return {Date} Local date
' @throws 10013 - ISO 8601 parsing error
''
Public Function ParseIso(utc_IsoString As String) As Date
    On Error GoTo utc_ErrorHandling

    Dim utc_Parts() As String
    Dim utc_DateParts() As String
    Dim utc_TimeParts() As String
    Dim utc_OffsetIndex As Long
    Dim utc_HasOffset As Boolean
    Dim utc_NegativeOffset As Boolean
    Dim utc_OffsetParts() As String
    Dim utc_Offset As Date

    utc_Parts = VBA.Split(utc_IsoString, "T")
    utc_DateParts = VBA.Split(utc_Parts(0), "-")
    ParseIso = VBA.DateSerial(VBA.CInt(utc_DateParts(0)), VBA.CInt(utc_DateParts(1)), VBA.CInt(utc_DateParts(2)))

    If UBound(utc_Parts) > 0 Then
        If VBA.InStr(utc_Parts(1), "Z") Then
            utc_TimeParts = VBA.Split(VBA.Replace(utc_Parts(1), "Z", ""), ":")
        Else
            utc_OffsetIndex = VBA.InStr(1, utc_Parts(1), "+")
            If utc_OffsetIndex = 0 Then
                utc_NegativeOffset = True
                utc_OffsetIndex = VBA.InStr(1, utc_Parts(1), "-")
            End If

            If utc_OffsetIndex > 0 Then
                utc_HasOffset = True
                utc_TimeParts = VBA.Split(VBA.Left$(utc_Parts(1), utc_OffsetIndex - 1), ":")
                utc_OffsetParts = VBA.Split(VBA.Right$(utc_Parts(1), Len(utc_Parts(1)) - utc_OffsetIndex), ":")

                Select Case UBound(utc_OffsetParts)
                Case 0
                    utc_Offset = TimeSerial(VBA.CInt(utc_OffsetParts(0)), 0, 0)
                Case 1
                    utc_Offset = TimeSerial(VBA.CInt(utc_OffsetParts(0)), VBA.CInt(utc_OffsetParts(1)), 0)
                Case 2
                    ' VBA.Val does not use regional settings, use for seconds to avoid decimal/comma issues
                    utc_Offset = TimeSerial(VBA.CInt(utc_OffsetParts(0)), VBA.CInt(utc_OffsetParts(1)), Int(VBA.Val(utc_OffsetParts(2))))
                End Select

                If utc_NegativeOffset Then: utc_Offset = -utc_Offset
            Else
                utc_TimeParts = VBA.Split(utc_Parts(1), ":")
            End If
        End If

        Select Case UBound(utc_TimeParts)
        Case 0
            ParseIso = ParseIso + VBA.TimeSerial(VBA.CInt(utc_TimeParts(0)), 0, 0)
        Case 1
            ParseIso = ParseIso + VBA.TimeSerial(VBA.CInt(utc_TimeParts(0)), VBA.CInt(utc_TimeParts(1)), 0)
        Case 2
            ' VBA.Val does not use regional settings, use for seconds to avoid decimal/comma issues
            ParseIso = ParseIso + VBA.TimeSerial(VBA.CInt(utc_TimeParts(0)), VBA.CInt(utc_TimeParts(1)), Int(VBA.Val(utc_TimeParts(2))))
        End Select

        ParseIso = ParseUtc(ParseIso)

        If utc_HasOffset Then
            ParseIso = ParseIso - utc_Offset
        End If
    End If

    Exit Function

utc_ErrorHandling:
    Err.Raise 10013, "UtcConverter.ParseIso", "ISO 8601 parsing error for " & utc_IsoString & ": " & Err.Number & " - " & Err.Description
End Function

Private Function utc_ConvertDate(utc_Value As Date, Optional utc_ConvertToUtc As Boolean = False) As Date
    Dim utc_ShellCommand As String
    Dim utc_Result As utc_ShellResult
    Dim utc_Parts() As String
    Dim utc_DateParts() As String

    If utc_ConvertToUtc Then
        utc_ShellCommand = "date -ur `date -jf '%Y-%m-%d %H:%M:%S' " & "'" & VBA.Format$(utc_Value, "yyyy-mm-dd HH:mm:ss") & "' " & " +'%s'` +'%Y-%m-%d %H:%M:%S'"
    Else
        utc_ShellCommand = "date -jf '%Y-%m-%d %H:%M:%S %z' " & "'" & VBA.Format$(utc_Value, "yyyy-mm-dd HH:mm:ss") & " +0000' " & "+'%Y-%m-%d %H:%M:%S'"
    End If

    utc_Result = utc_ExecuteInShell(utc_ShellCommand)

    If utc_Result.utc_Output = "" Then
        Err.Raise 10015, "UtcConverter.utc_ConvertDate", "'date' command failed"
    Else
        utc_Parts = Split(utc_Result.utc_Output, " ")
        utc_DateParts = Split(utc_Parts(0), "-")
        utc_TimeParts = Split(utc_Parts(1), ":")

        utc_ConvertDate = DateSerial(utc_DateParts(0), utc_DateParts(1), utc_DateParts(2)) + TimeSerial(utc_TimeParts(0), utc_TimeParts(1), utc_TimeParts(2))
    End If
End Function

Private Function utc_ExecuteInShell(utc_ShellCommand As String) As utc_ShellResult
#If VBA7 Then
    Dim utc_File As LongPtr
    Dim utc_Read As LongPtr
#Else
    Dim utc_File As Long
    Dim utc_Read As Long
#End If

    Dim utc_Chunk As String

    On Error GoTo utc_ErrorHandling
    utc_File = utc_popen(utc_ShellCommand, "r")

    If utc_File = 0 Then: Exit Function

    Do While utc_feof(utc_File) = 0
        utc_Chunk = VBA.Space$(50)
        utc_Read = CLng(utc_fread(utc_Chunk, 1, Len(utc_Chunk) - 1, utc_File))
        If utc_Read > 0 Then
            utc_Chunk = VBA.Left$(utc_Chunk, CLng(utc_Read))
            utc_ExecuteInShell.utc_Output = utc_ExecuteInShell.utc_Output & utc_Chunk
        End If
    Loop

utc_ErrorHandling:
    utc_ExecuteInShell.utc_ExitCode = CLng(utc_pclose(utc_File))
End Function

Public Function SetProperty(PropertyName As String, PropertySet As PropertyLocation, PropertyValue As Variant, Optional ContentLink As Boolean = False, Optional WhatWorkbook As Workbook) As Boolean

    
Dim WB As Workbook
Dim Prop As Office.DocumentProperty
Dim BProps As Office.DocumentProperties
Dim CProps As Office.DocumentProperties
Dim PropType As Variant

If PropertyName = vbNullString Then
    ''''''''''''''''''''''''''''''''''
    ' Don't allow blank names.
    ''''''''''''''''''''''''''''''''''
    SetProperty = False
    Exit Function
End If

If IsArray(PropertyValue) = True Then
    ''''''''''''''''''''''''''''''''''''''''''
    ' Don't allow arrays for a property value.
    ''''''''''''''''''''''''''''''''''''''''''
    SetProperty = False
    Exit Function
End If

'''''''''''''''''''''''''''''''''''''''''
' Set the workbook whose properties we
' will work with.
'''''''''''''''''''''''''''''''''''''''''
If WhatWorkbook Is Nothing Then
    Set WB = ThisWorkbook
Else
    Set WB = WhatWorkbook
End If
    
Set BProps = WB.BuiltinDocumentProperties
Set CProps = WB.CustomDocumentProperties

On Error Resume Next
''''''''''''''''''''''''''''''''''''''''''''
' If we are working with BuiltIn properties,
' we can simply assign the value since you
' can't link content to a BuiltIn property.
''''''''''''''''''''''''''''''''''''''''''''
If PropertySet = PropertyLocationBuiltIn Then
    Err.Clear
    Set Prop = BProps(PropertyName)
    If Err.Number = 0 Then
        '''''''''''''''''''''''''''''''''''
        ' Property exists. Set the value
        ' and get out.
        '''''''''''''''''''''''''''''''''''
        Prop.Value2 = PropertyValue
        SetProperty = True
        Exit Function
    End If
    '''''''''''''''''''''''''''''''''''
    ' Property doesn't exist. Get out.
    '''''''''''''''''''''''''''''''''''
    SetProperty = False
    Exit Function
End If

If (PropertySet = PropertyLocationCustom) Or (PropertySet = PropertyLocationBoth) Then
    ''''''''''''''''''''''''''''''''''''''''''''''''
    ' We need to delete the existing CustomProperty
    ' and replace it with a new CustomProperty with
    ' the same name. This allows us to change a
    ' LinkedContent property to an unlinked content
    ' property and vice-versa.
    ''''''''''''''''''''''''''''''''''''''''''''''''
    Err.Clear
    Set Prop = CProps(PropertyName)
    
    ''''''''''''''''''''''''''''''''''''''
    ' If the property exists, delete it.
    ''''''''''''''''''''''''''''''''''''''
    If Not Prop Is Nothing Then
        Prop.Delete
    End If
    Err.Clear
    If ContentLink = True Then
        ''''''''''''''''''''''''''''''''''''''''''''''''
        ' If ContentLink is True, then PropertyValue
        ' is the defined name to which the property
        ' will be linked. In this case, PropertyValue
        ' must be a String and the Name must exist.
        ''''''''''''''''''''''''''''''''''''''''''''''''
        
        
        If IsObject(PropertyValue) = True Then
            ''''''''''''''''''''''''''''''''''
            ' If PropertyValue is an Object,
            ' see if it is an Excel.Name.
            ''''''''''''''''''''''''''''''''''
            If TypeOf PropertyValue Is Excel.Name Then
                '''''''''''''''''''''''''''''''''''
                ' If it is a Name, set the link
                ' and get out.
                '''''''''''''''''''''''''''''''''''
                Err.Clear
                
                CProps.Add Name:=PropertyName, LinkToContent:=True, Type:=msoPropertyTypeString, LinkSource:=PropertyValue.Name
                SetProperty = (Err.Number = 0)
                Exit Function
            Else
                ''''''''''''''''''''''''''''''''''''''
                ' PropertyValue is an object but is
                ' not a Name. Get out.
                ''''''''''''''''''''''''''''''''''''''
                SetProperty = False
                Exit Function
            End If
        ElseIf VarType(PropertyValue) = vbString Then
            If NameExists(CStr(PropertyValue), WB) = False Then
                ''''''''''''''''''''''''''''''''
                ' Name doesn't exist. Get out.
                ''''''''''''''''''''''''''''''''
                SetProperty = False
                Exit Function
            End If
            '''''''''''''''''''''''''''''''
            ' Name exists. Set up the link.
            '''''''''''''''''''''''''''''''
            CProps.Add Name:=PropertyName, Type:=msoPropertyTypeString, LinkSource:=PropertyValue, LinkToContent:=True
                
        Else
            '''''''''''''''''''''''''''''''''
            ' PropertyValue is neither a Name
            ' nor a String. Get out.
            '''''''''''''''''''''''''''''''''
            SetProperty = False
            Exit Function
        End If
    Else
        ''''''''''''''''''''''''''''''''''''''''''
        ' We're not linking content. Just create
        ' the property, set the value, and get
        ' out.
        ''''''''''''''''''''''''''''''''''''''''''
        Err.Clear
        PropType = GetPropertyType(V:=PropertyValue)
        If IsNull(PropType) = True Then
            ''''''''''''''''''''''''''''''''''
            ' Illegal data type.
            ''''''''''''''''''''''''''''''''''
            SetProperty = False
            Exit Function
        End If
        Err.Clear
        CProps.Add Name:=PropertyName, LinkToContent:=False, Type:=PropType, Value:=PropertyValue
        SetProperty = (Err.Number = 0)
    End If
End If

''''''''''''''''''''''''''''''''
' If we get this far, success.
''''''''''''''''''''''''''''''''
SetProperty = True

End Function

Function GetProperty(PropertyName As String, PropertySet As PropertyLocation, Optional WhatWorkbook As Workbook) As Variant

Dim WB As Workbook
Dim Props1 As Office.DocumentProperties
Dim Props2 As Office.DocumentProperties
Dim Prop As Office.DocumentProperty

'''''''''''''''''''''''''''''''''''''''''
' Set the workbook whose properties we
' will search.
'''''''''''''''''''''''''''''''''''''''''
If WhatWorkbook Is Nothing Then
    Set WB = ThisWorkbook
Else
    Set WB = WhatWorkbook
End If


'''''''''''''''''''''''''''''''''''''''''
' Determine what property set we are
' going to look at.
'''''''''''''''''''''''''''''''''''''''''
Select Case PropertySet
    Case PropertyLocationBuiltIn
        Set Props1 = WB.BuiltinDocumentProperties
    Case PropertyLocationCustom
        Set Props1 = WB.CustomDocumentProperties
    Case PropertyLocationBoth
        Set Props1 = WB.BuiltinDocumentProperties
        Set Props2 = WB.CustomDocumentProperties
    Case Else
        GetProperty = CVErr(xlErrValue)
        Exit Function
End Select

On Error Resume Next
'''''''''''''''''''''''''''''''''''''''''
' Search either BuiltIn or Custom.
'''''''''''''''''''''''''''''''''''''''''
Set Prop = Props1(PropertyName)
If Err.Number <> 0 Then
    ''''''''''''''''''''''''''''''''''
    ' Not found in one set. See if
    ' we need to look in the other.
    ''''''''''''''''''''''''''''''''''
    If Not Props2 Is Nothing Then
        ''''''''''''''''''''''''''''''''''''
        ' We'll get here only if both Custom
        ' and BuiltIn properties are to be
        ' searched.
        ''''''''''''''''''''''''''''''''''''
        Err.Clear
        Set Prop = Props2(PropertyName)
        If Err.Number <> 0 Then
            ''''''''''''''''''''''''''''''''''''
            ' Property not found. Return NULL.
            ''''''''''''''''''''''''''''''''''''
            GetProperty = Null
            Exit Function
        End If
    Else
        ''''''''''''''''''''''''''''''''''''
        ' Property not found. Return NULL.
        ''''''''''''''''''''''''''''''''''''
        GetProperty = Null
        Exit Function
    End If
End If

''''''''''''''''''''''''''''''''''''
' Property found. Return the value.
''''''''''''''''''''''''''''''''''''
GetProperty = Prop.Value

End Function


Function PropertyExists(PropertyName As String, PropertySet As PropertyLocation, Optional WhatWorkbook As Workbook) As Boolean

PropertyExists = Not IsNull(GetProperty(PropertyName, PropertySet, WhatWorkbook))

End Function

Private Function NameExists(NameName As String, Optional WhatWorkbook As Workbook) As Boolean


Dim WB As Workbook
If WhatWorkbook Is Nothing Then
    Set WB = ThisWorkbook
Else
    Set WB = WhatWorkbook
End If

On Error Resume Next
NameExists = CBool(Len(WB.Names(NameName).Name))
End Function

Private Function GetPropertyType(V As Variant) As Variant


Select Case VarType(V)
    Case vbArray, vbDataObject, vbEmpty, vbError, vbNull, vbObject, vbUserDefinedType, vbCurrency, vbDecimal
        ''''''''''''''''''''''''''''''''''''
        ' Illegal types. Return NULL.
        ''''''''''''''''''''''''''''''''''''
        GetPropertyType = Null
        Exit Function
    ''''''''''''''''''''''''''''''''''
    ' All numeric types are rolled up
    ' into Floats. Strings and Booleans
    ' get their own types.
    ''''''''''''''''''''''''''''''''''
    Case vbString
        GetPropertyType = msoPropertyTypeString
    Case vbBoolean
        GetPropertyType = msoPropertyTypeBoolean
    Case Else
        GetPropertyType = msoPropertyTypeFloat
End Select

End Function
